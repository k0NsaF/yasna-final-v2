import hashlib

from fastapi import Depends, HTTPException, status

# Р”Р»СЏ РґРµРјРѕРЅСЃС‚СЂР°С†РёРё Р·Р°РґР°РЅ РІСЂРµРјРµРЅРЅС‹Р№ РїР°СЂРѕР»СЊ.
# РџРѕР·Р¶Рµ Р·Р°РјРµРЅРёРј РµРіРѕ РЅР° Р·РЅР°С‡РµРЅРёРµ РёР· .env.
MASTER_PASSWORD = os.getenv("MASTER_PASSWORD", "yasna-demo-2026")

# Р’СЂРµРјРµРЅРЅС‹Рµ СЃРµСЃСЃРёРё РјР°СЃС‚РµСЂР° С…СЂР°РЅСЏС‚СЃСЏ РІ РїР°РјСЏС‚Рё СЃРµСЂРІРµСЂР°.
# РЎСЂРѕРє РґРµР№СЃС‚РІРёСЏ РѕРґРЅРѕР№ СЃРµСЃСЃРёРё вЂ” 12 С‡Р°СЃРѕРІ.
MASTER_SESSION_SECONDS = 12 * 60 * 60
_master_sessions = {}

bearer_scheme = HTTPBearer(auto_error=False)


def hash_token(token: str) -> str:
    """Р’РѕР·РІСЂР°С‰Р°РµС‚ SHA-256-С…РµС€ С‚РѕРєРµРЅР°."""
    return hashlib.sha256(token.encode("utf-8")).hexdigest()


def generate_family_token() -> tuple[str, str]:
    """
    РЎРѕР·РґР°С‘С‚ СЃР»СѓС‡Р°Р№РЅС‹Р№ С‚РѕРєРµРЅ СЃРµРјСЊРё.
    Р’РѕР·РІСЂР°С‰Р°РµС‚ СЃР°Рј С‚РѕРєРµРЅ Рё РµРіРѕ С…РµС€.
    Р’ Р±Р°Р·Рµ СЃРѕС…СЂР°РЅСЏРµРј С‚РѕР»СЊРєРѕ С…РµС€.
    """
    token = secrets.token_urlsafe(32)
    token_hash = hash_token(token)
    return token, token_hash


def create_master_session(password: str) -> str:
    """РџСЂРѕРІРµСЂСЏРµС‚ РїР°СЂРѕР»СЊ РјР°СЃС‚РµСЂР° Рё СЃРѕР·РґР°С‘С‚ РІСЂРµРјРµРЅРЅСѓСЋ СЃРµСЃСЃРёСЋ."""
    if not hmac.compare_digest(password, MASTER_PASSWORD):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="РќРµРІРµСЂРЅС‹Р№ РїР°СЂРѕР»СЊ РјР°СЃС‚РµСЂР°",
        )

    token = secrets.token_urlsafe(32)
    _master_sessions[token] = time.time() + MASTER_SESSION_SECONDS
    return token


def get_current_master(
    credentials: HTTPAuthorizationCredentials | None = Depends(bearer_scheme),
) -> bool:
    """РџСЂРѕРІРµСЂСЏРµС‚ bearer-С‚РѕРєРµРЅ РјР°СЃС‚РµСЂР°."""
    if credentials is None:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="РўСЂРµР±СѓРµС‚СЃСЏ Р°РІС‚РѕСЂРёР·Р°С†РёСЏ РјР°СЃС‚РµСЂР°",
        )

    token = credentials.credentials
    expires_at = _master_sessions.get(token)

    if expires_at is None or expires_at < time.time():
        _master_sessions.pop(token, None)
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="РЎРµСЃСЃРёСЏ РјР°СЃС‚РµСЂР° РЅРµРґРµР№СЃС‚РІРёС‚РµР»СЊРЅР° РёР»Рё РёСЃС‚РµРєР»Р°",
        )

    return True


def get_current_family(
    credentials: HTTPAuthorizationCredentials | None = Depends(bearer_scheme),
    db: Session = Depends(get_db),
):
    """РќР°С…РѕРґРёС‚ СЃРµРјСЊСЋ РїРѕ Р°РєС‚РёРІРЅРѕРјСѓ РїРµСЂСЃРѕРЅР°Р»СЊРЅРѕРјСѓ С‚РѕРєРµРЅСѓ."""
    if credentials is None:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="РўСЂРµР±СѓРµС‚СЃСЏ РїРµСЂСЃРѕРЅР°Р»СЊРЅР°СЏ СЃСЃС‹Р»РєР° СЃРµРјСЊРё",
        )

    token_hash = hash_token(credentials.credentials)

    family_token = (
        db.query(FamilyToken)
        .filter(FamilyToken.token_hash == token_hash)
        .first()
    )

    if (

async function apiRequest(path, options = {}) {
    const headers = { ...options.headers };

    // Р”Р»СЏ РѕР±С‹С‡РЅС‹С… JSON-Р·Р°РїСЂРѕСЃРѕРІ СѓРєР°Р·С‹РІР°РµРј С‚РёРї РґР°РЅРЅС‹С….
    // Р”Р»СЏ FormData Р±СЂР°СѓР·РµСЂ СЃР°Рј СѓСЃС‚Р°РЅРѕРІРёС‚ РЅСѓР¶РЅС‹Р№ Content-Type.
    if (!(options.body instanceof FormData)) {
        headers["Content-Type"] =
            headers["Content-Type"] || "application/json";
    }

    const response = await fetch(API_URL + path, {
        ...options,
        headers
    });

    let data = null;

    const contentType = response.headers.get("content-type") || "";

    if (contentType.includes("application/json")) {
        data = await response.json();
    } else {
        data = await response.text();
    }

    if (!response.ok) {
        const message =
            typeof data === "object" && data !== null
                ? data.detail || "РћС€РёР±РєР° Р·Р°РїСЂРѕСЃР°"
                : data || "РћС€РёР±РєР° Р·Р°РїСЂРѕСЃР°";

        throw new Error(
            Array.isArray(message)
                ? message.map(item => item.msg).join(", ")
                : message
        );
    }

    return data;
}

async function downloadReceipt(receiptId) {
    try {
        const token = localStorage.getItem('family_token');

        if (!token) {
            throw new Error('РќРµС‚ РїРµСЂСЃРѕРЅР°Р»СЊРЅРѕРіРѕ С‚РѕРєРµРЅР° СЃРµРјСЊРё');
        }

        const response = await fetch(
            `${API_URL}/family/receipts/${receiptId}/download`,
            {
                headers: {
                    'Authorization': `Bearer ${token}`
                }
            }
        );

        if (!response.ok) {
            let message = 'РќРµ СѓРґР°Р»РѕСЃСЊ СЃРєР°С‡Р°С‚СЊ С‡РµРє';

            try {
                const data = await response.json();
                message = data.detail || message;
            } catch (_) {
                // РћСЃС‚Р°РІР»СЏРµРј СЃС‚Р°РЅРґР°СЂС‚РЅРѕРµ СЃРѕРѕР±С‰РµРЅРёРµ.
            }

            throw new Error(message);
        }

        const blob = await response.blob();
        const fileUrl = URL.createObjectURL(blob);

        const link = document.createElement('a');
        link.href = fileUrl;
        link.download = `receipt-${receiptId}`;

        document.body.appendChild(link);
        link.click();
        link.remove();

        setTimeout(() => URL.revokeObjectURL(fileUrl), 1000);

    } catch (err) {
        showToast(err.message, 'error');
    }
}

async function loadDashboard() {
    try {
        const [lessons, families, subscriptions] = await Promise.all([
            masterRequest('/master/lessons'),
            masterRequest('/master/families'),
            masterRequest('/master/subscriptions')
        ]);

        const today = new Date();

        const todayLessons = lessons.filter(lesson => {
            const date = new Date(lesson.starts_at);

            return date.getFullYear() === today.getFullYear()
                && date.getMonth() === today.getMonth()
                && date.getDate() === today.getDate();
        });

        const childrenTotal = families.reduce((sum, family) => {
            return sum + (family.children?.length || 0);
        }, 0);

        const endingSubs = subscriptions.filter(sub => {
            if (!sub.is_active) return false;

            const remaining = sub.total_lessons - sub.used_lessons;

            const expiresSoon = sub.expires_at &&
                new Date(sub.expires_at) <=
                new Date(Date.now() + 7 * 24 * 60 * 60 * 1000);

            return remaining <= 2 || expiresSoon;
        });

        document.getElementById('stat-lessons').innerText =
            todayLessons.length;

        document.getElementById('stat-children').innerText =
            childrenTotal;

        document.getElementById('stat-ending').innerText =
            endingSubs.length;

    } catch (e) {
        showToast('РћС€РёР±РєР° Р·Р°РіСЂСѓР·РєРё РіР»Р°РІРЅРѕР№ РїР°РЅРµР»Рё: ' + e.message, 'error');
    }
}

    if (!isoString) return '';
